cmake_minimum_required(VERSION 3.25)
project(tt_transfer_runtime VERSION 0.1.0 LANGUAGES CXX)
option(TT_TRANSFER_WITH_UMD "Build real pinned TT-UMD integration (Linux)" OFF)
set(TT_TRANSFER_SANITIZER "" CACHE STRING "address or thread; empty disables")
find_package(Threads REQUIRED)
add_library(tt_transfer src/runtime.cpp src/cpu_backend.cpp)
target_include_directories(tt_transfer PUBLIC include)
target_compile_features(tt_transfer PUBLIC cxx_std_20)
target_link_libraries(tt_transfer PUBLIC Threads::Threads)
target_compile_options(tt_transfer PRIVATE -Wall -Wextra -Wpedantic -Werror)
if(TT_TRANSFER_SANITIZER STREQUAL "address")
  target_compile_options(tt_transfer PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(tt_transfer PUBLIC -fsanitize=address,undefined)
elseif(TT_TRANSFER_SANITIZER STREQUAL "thread")
  target_compile_options(tt_transfer PUBLIC -fsanitize=thread -fno-omit-frame-pointer)
  target_link_options(tt_transfer PUBLIC -fsanitize=thread)
elseif(NOT TT_TRANSFER_SANITIZER STREQUAL "")
  message(FATAL_ERROR "Unknown sanitizer: ${TT_TRANSFER_SANITIZER}")
endif()
enable_testing()
add_executable(runtime_tests tests/runtime_tests.cpp)
target_link_libraries(runtime_tests PRIVATE tt_transfer)
target_compile_options(runtime_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_test(NAME runtime_contract COMMAND runtime_tests)
set_tests_properties(runtime_contract PROPERTIES TIMEOUT 30)
add_executable(transfer_demo src/demo.cpp)
target_link_libraries(transfer_demo PRIVATE tt_transfer)
add_executable(cpu_benchmark benchmarks/cpu_benchmark.cpp)
target_link_libraries(cpu_benchmark PRIVATE tt_transfer)
if(TT_TRANSFER_WITH_UMD)
  include(cmake/Umd.cmake)
endif()
